RODAR O SITE
    python manage.py run  # ativa o servidor do sistema

PARA QUALQUER ALTERAÇÃO NOS MODELOS DO BANCO DE DADOS
    python manage.py makemigrations  #criar um banco de dados baseado na classe Model
    python manage.py migrate

CRIAÇÃO DE USUÁRIO
    python manage.py createsuperuser  # cria um superusuario na rota admin do django
